cmake_minimum_required(VERSION 3.20)
project(corelib_package_consumer LANGUAGES C CXX)
option(CORELIB_CONSUMER_BUILD_CPP "Build the C++ package consumer" OFF)
option(CORELIB_CONSUMER_BUILD_GATEWAY "Build gateway package consumers" OFF)
find_package(Corelib 1 CONFIG REQUIRED)
add_executable(package_consumer main.c)
target_link_libraries(package_consumer PRIVATE Corelib::Device)

if(CORELIB_CONSUMER_BUILD_CPP)
  find_package(Corelib 1 CONFIG REQUIRED COMPONENTS Cpp)
  add_executable(package_consumer_cpp main.cpp)
  target_link_libraries(package_consumer_cpp PRIVATE Corelib::DeviceCpp)
endif()

if(CORELIB_CONSUMER_BUILD_GATEWAY)
  find_package(Corelib 1 CONFIG REQUIRED COMPONENTS Gateway)
  add_executable(package_consumer_gateway main_gateway.c)
  target_link_libraries(package_consumer_gateway PRIVATE Corelib::Gateway)
  if(CORELIB_CONSUMER_BUILD_CPP)
    find_package(Corelib 1 CONFIG REQUIRED COMPONENTS GatewayCpp)
    add_executable(package_consumer_gateway_cpp main_gateway.cpp)
    target_link_libraries(package_consumer_gateway_cpp PRIVATE Corelib::GatewayCpp)
  endif()
endif()
